```python
class IndustrialDataPipeline:
    def __init__(self, data_path: str):
        self.data_path = data_path
        self.train_data = None
        self.test_data = None
    
    def load_and_clean(self) -> pd.DataFrame:
        # Multi-year production data
        df = pd.read_csv(self.data_path)
        df = self.remove_outliers(df)
        df = self.normalize_features(df)
        return df
    
    def split_temporal(self, df: pd.DataFrame, split_date: str):
        # Time-based split for production data
        self.train_data = df[df['date'] < split_date]
        self.test_data = df[df['date'] >= split_date]
```

**Exercise 2.1 - Federated Multi-Site:**
```python
class MultiSiteOptimizer:
    def __init__(self, sites: List[str]):
        self.sites = sites
        self.site_agents = {}
    
    def federated_optimize(self, max_iterations: int = 100):
        # Initialize agents for each site
        for site in self.sites:
            self.site_agents[site] = OptimizationAgent(site)
        
        # Federated averaging loop
        for iteration in range(max_iterations):
            # Local optimization at each site
            local_results = [agent.optimize_local() 
                           for agent in self.site_agents.values()]
            
            # Aggregate results (FedAvg)
            global_best = self.aggregate_results(local_results)
            
            # Broadcast global best
            for agent in self.site_agents.values():
                agent.update_from_global(global_best)
```
